# Dependency scan
python scanner.py scan requirements.txt -o findings.json --html report.html

# URL DAST scan
python scanner.py scan https://staging.example.com --severity critical,high

# Probe with active PoCs
python scanner.py probe findings.json --target https://staging.example.com \
    -o verified.json --html verified.html

# Probe + fingerprint real host libraries
python scanner.py probe findings.json --target https://staging.example.com \
    --fingerprint \
    -o verified.json --html verified.html

# Authenticated, WAF-aware, rate-limited
python scanner.py probe findings.json --target https://app.example.com \
    --header "Authorization: Bearer $TOKEN" \
    --cookie "session=abc123" \
    --basic-auth admin:pass \
    --waf-bypass --profile stealth --insecure \
    --fingerprint \
    -o verified.json --html verified.html

# Single CVE lookup
python scanner.py enrich CVE-2024-3400

# Inspect libraries
python scanner.py --list-cves       # 45 entries
python scanner.py --list-payloads   # 14 CWE probes
