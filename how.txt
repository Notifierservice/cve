# Scan dependencies
python scanner.py scan requirements.txt -o findings.json --html report.html

# Scan npm deps
python scanner.py scan package.json -o findings.json

# Scan URL via Nuclei DAST
python scanner.py scan https://staging.example.com -o dast.json

# Probe findings against live target
python scanner.py probe findings.json --target https://staging.example.com \
    --header "Authorization: Bearer $TOKEN" -o verified.json --html verified.html

# Dry-run (no payloads sent)
python scanner.py probe findings.json --target https://staging.example.com --dry-run

# Authenticated probe with WAF evasion
python scanner.py probe findings.json --target https://app.example.com \
    --basic-auth admin:pass --cookie "session=abc" --waf-bypass \
    --profile stealth --insecure -o verified.json

# Look up one CVE
python scanner.py enrich CVE-2024-3400

# Inspect libraries
python scanner.py --list-cves
python scanner.py --list-payloads
